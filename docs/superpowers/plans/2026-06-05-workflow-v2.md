# Workflow V2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Introduce 3 new roles (designer, coordinator, production), remove the approval workflow in favour of direct sales→schedule push with FCFS clash detection, add a Scheduler FCFS Board tab, support installer suggestions by sales, formal assignment by coordinator/scheduler, external installer temp links, sub-installer relationships, and a task list bucket.

**Architecture:** Built in 4 independent phases on the `feat-workflow-v2` branch — each phase ships testable software. Phase 1 (roles + workflow) is a prerequisite for all others. Phases 2–4 can be worked in any order after Phase 1. No new npm packages required. All DB changes land in numbered Supabase migrations.

**Tech Stack:** Next.js 15 (App Router), TypeScript strict, Tailwind, Supabase (Postgres + RLS + Realtime), Cloudflare R2, Anthropic Claude, Telegram Bot API.

**Assumptions documented here (confirm with Nic if unsure):**
- Coordinator also has access to the FCFS Board tab (they assign installers, need to see clashes)
- Designer and Production can see all scheduled jobs in the Schedule list (view only)
- Production role users see schedule list but can only edit their specific fields in job form
- "awaiting_approval" is kept in the DB enum but no new jobs will enter that state (for historical data safety)
- A job can have multiple coordinators (via existing `job_coordinators` join table from migration 0029)
- External installer temp link is generated manually by coordinator/scheduler from within the job form

- [ ] **Step 3: Create suggest-installer API route**

```typescript
// src/app/api/jobs/[id]/suggest-installer/route.ts
import { createRouteHandlerClient } from '@/lib/supabase/server'
import { NextRequest, NextResponse } from 'next/server'

export async function POST(req: NextRequest, { params }: { params: { id: string } }) {
  const supabase = createRouteHandlerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { data: profile } = await supabase
    .from('users').select('id, role').eq('auth_id', user.id).single()
  if (profile?.role !== 'sales')
    return NextResponse.json({ error: 'Forbidden' }, { status: 403 })

  const { user_id, action } = await req.json()
  const jobId = params.id

  if (action === 'add') {
    await supabase.from('job_assignees').upsert({
      job_id:       jobId,
      user_id,
      is_suggestion: true,
      suggested_by:  profile.id,
    }, { onConflict: 'job_id,user_id' })
  } else {
    await supabase.from('job_assignees')
      .delete()
      .eq('job_id', jobId)
      .eq('user_id', user_id)
      .eq('is_suggestion', true)
  }

  return NextResponse.json({ ok: true })
}
```

export async function getFCFSWeek(
  supabase: SupabaseClient,
  weekStart: string, // YYYY-MM-DD (Monday)
  weekEnd: string    // YYYY-MM-DD (Sunday)

    .from('jobs')

        users(name))
    `)

    .lte('date', weekEnd)

    id:            job.id,
    project_title: job.project_title,
    client:        job.client,
    date:          job.date,
    time_start:    job.time_start,
    time_end:      job.time_end,
    fcfs_rank:     idx + 1,
    assignees:     (job.job_assignees as any[]).map(a => ({
      user_id:          a.user_id,
      name:             a.users?.name ?? 'Unknown',
      is_suggestion:    a.is_suggestion,
      is_sub_installer: a.is_sub_installer,
    })),

// src/lib/utils/clashDetection.ts
import { FCFSJob } from '@/lib/supabase/queries/fcfs'

export interface Clash {
  installerId:   string
  installerName: string
  date:          string
  jobA:          FCFSJob
  jobB:          FCFSJob
}

function toMinutes(time: string): number {
  const [h, m] = time.split(':').map(Number)
  return h * 60 + m
}

function overlaps(a: FCFSJob, b: FCFSJob): boolean {
  if (!a.time_end || !b.time_end) return false
  const aStart = toMinutes(a.time_start)
  const aEnd   = toMinutes(a.time_end)
  const bStart = toMinutes(b.time_start)
  const bEnd   = toMinutes(b.time_end)
  return aStart < bEnd && bStart < aEnd
}

export function detectClashes(jobs: FCFSJob[]): Clash[] {
  const clashes: Clash[] = []

  for (let i = 0; i < jobs.length; i++) {
    for (let j = i + 1; j < jobs.length; j++) {
      const a = jobs[i]
      const b = jobs[j]
      if (a.date !== b.date) continue
      if (!overlaps(a, b)) continue

      // Find shared formal assignees (non-suggestion, non-sub)
      const aAssigned = a.assignees.filter(x => !x.is_suggestion && !x.is_sub_installer)
      const bAssigned = b.assignees.filter(x => !x.is_suggestion && !x.is_sub_installer)

      for (const aA of aAssigned) {
        const shared = bAssigned.find(bA => bA.user_id === aA.user_id)
        if (shared) {
          clashes.push({
            installerId:   aA.user_id,
            installerName: aA.name,
            date:          a.date,
            jobA:          a.fcfs_rank <= b.fcfs_rank ? a : b,
            jobB:          a.fcfs_rank <= b.fcfs_rank ? b : a,
          })
        }
      }
    }
  }

  return clashes
}

  const clashes = (conflicts ?? []).filter((c: any) => {
    const j = c.jobs
    if (j.date !== currentJob?.date) return false
    if (!j.time_end || !currentJob.time_end) return false
    const aS = toMinutes(currentJob.time_start), aE = toMinutes(currentJob.time_end)
    const bS = toMinutes(j.time_start),           bE = toMinutes(j.time_end)
    return aS < bE && bS < aE
  })

  return NextResponse.json({ hasClash: clashes.length > 0, clashes })
}

      method: 'POST', headers: { 'Content-Type': 'application/json' },

// src/app/api/jobs/[id]/external-installers/route.ts
import { createRouteHandlerClient } from '@/lib/supabase/server'
import { NextRequest, NextResponse } from 'next/server'

const ALLOWED_ROLES = ['sales', 'scheduler', 'coordinator', 'admin']

// GET — fetch all external installer POCs for this job
export async function GET(req: NextRequest, { params }: { params: { id: string } }) {
  const supabase = createRouteHandlerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { data } = await supabase
    .from('job_external_installers')
    .select('*')
    .eq('job_id', params.id)
    .order('sort_order')

  return NextResponse.json(data ?? [])
}

// POST — add a new external POC row
export async function POST(req: NextRequest, { params }: { params: { id: string } }) {
  const supabase = createRouteHandlerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { data: profile } = await supabase
    .from('users').select('role').eq('auth_id', user.id).single()
  if (!ALLOWED_ROLES.includes(profile?.role ?? ''))
    return NextResponse.json({ error: 'Forbidden' }, { status: 403 })

  // Get current max sort_order for this job
  const { data: last } = await supabase
    .from('job_external_installers')
    .select('sort_order')
    .eq('job_id', params.id)
    .order('sort_order', { ascending: false })
    .limit(1)
    .single()

  const { data: row } = await supabase
    .from('job_external_installers')
    .insert({
      job_id:     params.id,
      poc_name:   '',
      contact:    '',
      quantity:   1,
      sort_order: (last?.sort_order ?? -1) + 1,
    })
    .select()
    .single()

  return NextResponse.json(row)
}

// PATCH — update a single POC row's fields
export async function PATCH(req: NextRequest, { params }: { params: { id: string } }) {
  const supabase = createRouteHandlerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { data: profile } = await supabase
    .from('users').select('role').eq('auth_id', user.id).single()
  if (!ALLOWED_ROLES.includes(profile?.role ?? ''))
    return NextResponse.json({ error: 'Forbidden' }, { status: 403 })

  const { row_id, poc_name, contact, quantity } = await req.json()
  await supabase
    .from('job_external_installers')
    .update({ poc_name, contact, quantity })
    .eq('id', row_id)
    .eq('job_id', params.id) // safety: only update rows belonging to this job

  return NextResponse.json({ ok: true })
}

// DELETE — remove a POC row
export async function DELETE(req: NextRequest, { params }: { params: { id: string } }) {
  const supabase = createRouteHandlerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { data: profile } = await supabase
    .from('users').select('role').eq('auth_id', user.id).single()
  if (!ALLOWED_ROLES.includes(profile?.role ?? ''))
    return NextResponse.json({ error: 'Forbidden' }, { status: 403 })

  const { row_id } = await req.json()
  await supabase
    .from('job_external_installers')
    .delete()
    .eq('id', row_id)
    .eq('job_id', params.id)

  return NextResponse.json({ ok: true })
}

// src/features/job-detail/ExternalInstallerBucket.tsx
'use client'
import { useEffect, useState } from 'react'
import { Role } from '@/lib/supabase/types'

interface ExternalPOC {
  id:        string
  poc_name:  string
  contact:   string
  quantity:  number
}

interface ExternalInstallerBucketProps {
  jobId: string
  role:  Role
}

export default function ExternalInstallerBucket({ jobId, role }: ExternalInstallerBucketProps) {
  const [rows, setRows] = useState<ExternalPOC[]>([])
  const canEdit = (['sales','scheduler','coordinator','admin'] as Role[]).includes(role)

  useEffect(() => {
    fetch(`/api/jobs/${jobId}/external-installers`)
      .then(r => r.json())
      .then(setRows)
  }, [jobId])

  const addRow = async () => {
    const res = await fetch(`/api/jobs/${jobId}/external-installers`, { method: 'POST' })
    const row = await res.json()
    setRows(prev => [...prev, row])
  }

  const updateRow = async (id: string, field: keyof ExternalPOC, value: string | number) => {
    setRows(prev => prev.map(r => r.id === id ? { ...r, [field]: value } : r))
    await fetch(`/api/jobs/${jobId}/external-installers`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        row_id: id,
        ...rows.find(r => r.id === id),
        [field]: value,
      }),
    })
  }

  const removeRow = async (id: string) => {
    setRows(prev => prev.filter(r => r.id !== id))
    await fetch(`/api/jobs/${jobId}/external-installers`, {
      method: 'DELETE',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ row_id: id }),
    })
  }

  return (
    <div className="mt-3 rounded-xl border border-amber-200 bg-amber-50 p-4 space-y-3">
      <p className="text-xs font-semibold text-amber-700 uppercase tracking-wide">External Installers</p>

      {rows.map((row, idx) => (
        <div key={row.id} className="rounded-lg border border-amber-200 bg-paper p-3 space-y-2">
          <div className="flex items-center justify-between mb-1">
            <span className="text-xs text-muted">Team {idx + 1}</span>
            {canEdit && rows.length > 1 && (
              <button
                onClick={() => removeRow(row.id)}
                className="text-xs text-muted hover:text-terracotta"
              >
                Remove
              </button>
            )}
          </div>

          <input
            className="w-full text-sm border border-line rounded-lg px-3 py-1.5 bg-bg text-ink"
            placeholder="POC Name"
            value={row.poc_name}
            disabled={!canEdit}
            onChange={e => updateRow(row.id, 'poc_name', e.target.value)}
          />
          <input
            className="w-full text-sm border border-line rounded-lg px-3 py-1.5 bg-bg text-ink"
            placeholder="Contact Number"
            value={row.contact}
            disabled={!canEdit}
            onChange={e => updateRow(row.id, 'contact', e.target.value)}
          />
          <div className="flex items-center gap-2">
            <label className="text-xs text-muted whitespace-nowrap">No. of Installers</label>
            <input
              type="number"
              min={1}
              className="w-20 text-sm border border-line rounded-lg px-3 py-1.5 bg-bg text-ink"
              value={row.quantity}
              disabled={!canEdit}
              onChange={e => updateRow(row.id, 'quantity', parseInt(e.target.value) || 1)}
            />
          </div>
        </div>
      ))}

      {canEdit && (
        <button
          onClick={addRow}
          className="w-full text-sm border border-dashed border-amber-300 rounded-lg py-2 text-amber-700 hover:bg-amber-100 transition-colors"
        >
          + Add another POC
        </button>
      )}
    </div>
  )
}
